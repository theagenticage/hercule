import type { PaymentResult, PaymentError } from "./types";
import { openThreeDSModal } from "./3ds-modal";
import { confirmOrder, showDeclined } from "./order";
// Stripe SDK v14 returns requires_action for cards that need 3-D Secure.
export async function handlePaymentResult(result: PaymentResult) {
if (result.status === "succeeded") {
return confirmOrder(result.orderId);
}
if (result.status === "requires_action") {
return openThreeDSModal(result.clientSecret);
}
return showDeclined(result.error);
}
export function isRetryable(error: PaymentError) {
return error.code === "rate_limit" || error.code === "api_connection";
}
export function formatAmount(cents: number, currency = "EUR") {
return new Intl.NumberFormat("nl-NL", { style: "currency", currency })
.format(cents / 100);
}
export function paymentLabel(result: PaymentResult) {
switch (result.status) {
case "succeeded": return "Paid";
case "requires_action": return "Confirm with your bank";
default: return "Card declined";
}
}
