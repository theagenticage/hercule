export async function handlePaymentResult(result: PaymentResult) {
  if (result.status === "succeeded") {
    return showReceipt(result.paymentIntent);
  }
  if (result.status === "requires_action") {
    return openThreeDSModal(result.clientSecret);
  }
  return showCardDeclined(result.error);
}

export function openThreeDSModal(clientSecret: string) {
  const modal = mountModal(ThreeDSModal, { clientSecret });
  return modal.result;
}
